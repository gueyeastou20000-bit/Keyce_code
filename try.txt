+    {
+        TextDocument? document = Document;
+        if (document is null)
+            return;

+        IReadOnlyList<Breakpoint> breakpoints = _store.For(FilePath);

+        foreach (Breakpoint removed in _anchors.Keys.Where(b => !breakpoints.Contains(b)).ToList())
+            _anchors.Remove(removed);

+            if (_anchors.ContainsKey(breakpoint))
+                continue;

+    {
+        if (FilePath is not { } file || Document is not { } document)

+        int previousLength = document.TextLength - e.InsertionLength + e.RemovalLength;
+        if (e.Offset == 0 && e.RemovalLength > 0 && e.RemovalLength == previousLength)
+        {
+            _anchors.Clear();
+            SyncAnchors();
+            InvalidateVisual();
+            return;
+        }
+
+        bool moved = false;
+
+        foreach ((Breakpoint breakpoint, TextAnchor anchor) in _anchors)
+        {
+            if (anchor.Line != breakpoint.Line)
+            {
+                breakpoint.Line = anchor.Line;
+                moved = true;
+            }
+        }
+
+        if (moved)
+            _store.NotifyMoved(file);
+    }

+/// <summary>Tous les points d'arrêt, rangés par fichier. Partagé par les éditeurs, la vue Débogage et la session.</summary>
+public sealed class BreakpointStore
+{
+    private readonly Dictionary<string, List<Breakpoint>> _byFile = new(StringComparer.OrdinalIgnoreCase);
+
+    /// <summary>Les points d'arrêt d'un fichier ont changé (paramètre : chemin complet du fichier).</summary>
+    public event Action<string>? Changed;
+
+    public IEnumerable<Breakpoint> All =>
+        _byFile.Values.SelectMany(list => list).OrderBy(b => b.FilePath).ThenBy(b => b.Line);
+
+    public IReadOnlyList<Breakpoint> For(string? filePath)
+    {
+        if (filePath is null)
+            return [];
+
+        return _byFile.TryGetValue(Normalize(filePath), out List<Breakpoint>? list) ? list : [];
+    }
+
+    public Breakpoint? Find(string filePath, int line) =>
+        For(filePath).FirstOrDefault(b => b.Line == line);
+
+    /// <summary>Ajoute un point d'arrêt sur la ligne, ou le retire s'il existe déjà (comme F9 dans VS Code).</summary>
+    public void Toggle(string filePath, int line)
+    {
+        string file = Normalize(filePath);
+        Breakpoint? existing = Find(file, line);
+
+        if (existing is not null)
+        {
+            Remove(existing);
+            return;
+        }
+
+        if (!_byFile.TryGetValue(file, out List<Breakpoint>? list))
+            _byFile[file] = list = new List<Breakpoint>();
+
+        list.Add(new Breakpoint(file, line));
+        Changed?.Invoke(file);
+    }
+
+    public void Remove(Breakpoint breakpoint)
+    {
+        if (_byFile.TryGetValue(breakpoint.FilePath, out List<Breakpoint>? list) && list.Remove(breakpoint))
+        {
+            if (list.Count == 0)
+                _byFile.Remove(breakpoint.FilePath);
+
+            Changed?.Invoke(breakpoint.FilePath);
+        }
+    }
+
+    public void RemoveAll()
+    {
+        List<string> files = _byFile.Keys.ToList();
+        _byFile.Clear();
+
+        foreach (string file in files)
+            Changed?.Invoke(file);
+    }
+
+    public void SetEnabled(Breakpoint breakpoint, bool enabled)
+    {
+        breakpoint.Enabled = enabled;
+        Changed?.Invoke(breakpoint.FilePath);
+    }
+
+    public void SetCondition(Breakpoint breakpoint, string? condition)
+    {
+        breakpoint.Condition = string.IsNullOrWhiteSpace(condition) ? null : condition.Trim();
+        Changed?.Invoke(breakpoint.FilePath);
+    }
+
+    /// <summary>
+    /// Les lignes ont bougé (texte modifié) : on retire les doublons et on prévient.
+    /// Appelé par la marge de l'éditeur.
+    /// </summary>
+    public void NotifyMoved(string filePath)
+    {
+        string file = Normalize(filePath);
+        if (!_byFile.TryGetValue(file, out List<Breakpoint>? list))
+            return;
+
+        List<Breakpoint> unique = list.GroupBy(b => b.Line).Select(g => g.First()).ToList();
+        list.Clear();
+        list.AddRange(unique);
+
+        Changed?.Invoke(file);
+    }
+
+    /// <summary>Points d'arrêt actifs d'un fichier, au format attendu par la session de débogage.</summary>
+    public IReadOnlyList<SourceBreakpoint> ToSource(string filePath) =>
+        For(filePath)
+            .Where(b => b.Enabled)
+            .Select(b => new SourceBreakpoint(b.Line, b.Condition))
+            .ToList();
+
+    /// <summary>Tous les points d'arrêt actifs, par fichier.</summary>
+    public IReadOnlyDictionary<string, IReadOnlyList<SourceBreakpoint>> ToSourceByFile() =>
+        _byFile.Keys.ToDictionary(file => file, file => ToSource(file), StringComparer.OrdinalIgnoreCase);
+
+    public static string Normalize(string filePath) => Path.GetFullPath(filePath);

+    private async Task RefreshWatchesAsync()
+    {
+        if (_session is not { State: DebugState.Paused } session)
+        {
+            _view.RenderWatches(null);
+            return;
+        }
+
+        var results = new List<(string Text, bool IsError)>();
+
+        foreach (string expression in _view.Watches.ToList())
+        {
+            try
+            {
+                DebugEvaluation evaluation = await session.EvaluateAsync(expression, _currentFrame);
+                results.Add(evaluation.Failed ? (evaluation.Error!, true) : (evaluation.Result, false));
+            }
+            catch (Exception ex)
+            {
+                results.Add((ex.Message, true));
+            }
+        }

+    private async Task OnConsoleSubmittedAsync(string text)
+    {
+        if (_session is not { } session)
+        {
+            _console.WriteInfo("Démarrez le débogage (F5) pour utiliser la console.");
+            return;
+        }
+
+        // Programme en cours : le texte est envoyé à input()
+        if (session.State != DebugState.Paused)
+        {
+            _console.WriteInfo(text);
+            session.SendInput(text);
+            return;
+        }
+
+        // En pause : on évalue l'expression à l'endroit de l'arrêt
+        if (text.Trim().Length == 0)
+            return;
+
+        _console.WriteInfo("› " + text);
+
+        try
+        {
+            DebugEvaluation evaluation = await session.EvaluateAsync(text, _currentFrame);
+
+            if (evaluation.Failed)
+                _console.WriteError(evaluation.Error!);
+            else if (evaluation.Result.Length > 0)
+                _console.WriteResult(evaluation.Result);
+
+            // Une instruction (ex. x = 5) a pu changer les variables
+            await LoadScopesAsync(_currentFrame);
+            await RefreshWatchesAsync();
+        }
+        catch (Exception ex)
+        {
+            _console.WriteError(ex.Message);
+        }
+    }

+    // =====================================================================
+    // Points d'arrêt pendant le débogage
+    // =====================================================================
+
+    private async Task SyncBreakpointsAsync(string file)
+    {
+        if (_session is not { State: DebugState.Running or DebugState.Paused } session)
+            return;
+
+        try
+        {
+            IReadOnlyList<int> accepted = await session.SetBreakpointsAsync(file, _store.ToSource(file));
+
+            foreach (Breakpoint breakpoint in _store.For(file))
+                breakpoint.Verified = !breakpoint.Enabled || accepted.Contains(breakpoint.Line);
+
+            foreach (BreakpointMargin margin in _margins.Values)
+                margin.InvalidateVisual();
+        }
+        catch (Exception ex)
+        {
+            _console.WriteError($"Points d'arrêt non transmis : {ex.Message}");
+        }
+    }
+
+    // =====================================================================
+    // Aller à une ligne
+    // =====================================================================
+
+    private async Task ShowFrameLocationAsync(DebugFrame frame, bool isTopFrame)
+    {
+        // Document « Sans titre » exécuté depuis une copie temporaire : pas d'onglet à ouvrir
+        if (frame.FilePath.StartsWith(TempRoot, StringComparison.OrdinalIgnoreCase))
+        {
+            _console.WriteInfo($"Arrêt ligne {frame.Line} (enregistrez le fichier pour voir la ligne dans l'éditeur).");
+            return;
+        }
+
+        if (!File.Exists(frame.FilePath))
+            return;
+
+        _location.Set(frame.FilePath, frame.Line, isTopFrame);
+        await RevealAsync(frame.FilePath, frame.Line);
+    }
+
+    /// <summary>Ouvre le fichier (si besoin), place le curseur sur la ligne et la fait défiler à l'écran.</summary>
+    private async Task RevealAsync(string file, int line)
+    {
+        await _files.OpenPathAsync(file);
+
+        EditorTab? tab = _tabs.FindByPath(file);
+        if (tab is null)
+            return;
+
+        _tabs.Activate(tab);
+        AttachEditors();
+
+        TextEditor editor = tab.Editor.InnerEditor;
+        _ = editor.Dispatcher.BeginInvoke(DispatcherPriority.Loaded, new Action(() =>
+        {
+            if (line < 1 || line > editor.Document.LineCount)
+                return;
+
+            editor.TextArea.Caret.Location = new TextLocation(line, 1);
+            editor.ScrollToLine(line);
+            editor.TextArea.TextView.InvalidateLayer(KnownLayer.Background);
+        }));
+    }

+{

+/// <summary>
+/// Décode le contenu d'un PDF avec la bibliothèque PdfPig :
+/// texte des pages, recherche, informations du document.
+/// </summary>
+public static class PdfTextReader
+{
+    private const int MaxHits = 300;
+
+    /// <summary>Texte d'une page, dans l'ordre de lecture.</summary>
+    public static Task<string> GetPageTextAsync(string path, string? password, int pageIndex) =>
+        Task.Run(() =>
+        {
+            using PdfDocument document = Open(path, password);
+            Page page = document.GetPage(pageIndex + 1); // PdfPig compte les pages à partir de 1
+            return ContentOrderTextExtractor.GetText(page);
+        });
+
+    /// <summary>Cherche un texte dans toutes les pages (sans tenir compte des majuscules).</summary>
+    public static Task<IReadOnlyList<PdfSearchHit>> SearchAsync(
+        string path, string? password, string query, CancellationToken cancellationToken) =>
+        Task.Run<IReadOnlyList<PdfSearchHit>>(() =>
+        {
+            var hits = new List<PdfSearchHit>();
+            using PdfDocument document = Open(path, password);
+
+            foreach (Page page in document.GetPages())
+            {
+                cancellationToken.ThrowIfCancellationRequested();
+
+                string text = ContentOrderTextExtractor.GetText(page).Replace('\r', ' ').Replace('\n', ' ');
+                int index = text.IndexOf(query, StringComparison.CurrentCultureIgnoreCase);
+
+                while (index >= 0 && hits.Count < MaxHits)
+                {
+                    hits.Add(new PdfSearchHit(page.Number - 1, Excerpt(text, index, query.Length)));
+                    index = text.IndexOf(query, index + query.Length, StringComparison.CurrentCultureIgnoreCase);
+                }
+
+                if (hits.Count >= MaxHits)
+                    break;
+            }
+
+            return hits;
+        }, cancellationToken);
+
+    /// <summary>Informations générales : pages, version, titre, auteur, logiciel de création...</summary>
+    public static Task<PdfInfo> GetInfoAsync(string path, string? password) =>
+        Task.Run(() =>
+        {
+            using PdfDocument document = Open(path, password);
+
+            return new PdfInfo(
+                document.NumberOfPages,
+                document.Version.ToString(CultureInfo.InvariantCulture),
+                Clean(document.Information.Title),
+                Clean(document.Information.Author),
+                Clean(document.Information.Creator),
+                Clean(document.Information.Producer),
+                document.IsEncrypted,
+                new FileInfo(path).Length);
+        });
+
+    private static PdfDocument Open(string path, string? password)
+    {
+        if (string.IsNullOrEmpty(password))
+            return PdfDocument.Open(path);
+
+        return PdfDocument.Open(path, new ParsingOptions { Passwords = new List<string> { password } });
+    }
+
+    private static string Excerpt(string text, int index, int length)
+    {
+        int start = Math.Max(0, index - 35);
+        int end = Math.Min(text.Length, index + length + 35);
+        string excerpt = text[start..end].Trim();
+
+        return (start > 0 ? "…" : "") + excerpt + (end < text.Length ? "…" : "");
+    }
+
+    private static string? Clean(string? value) =>
+        string.IsNullOrWhiteSpace(value) ? null : value.Trim();
+}

+    public static bool IsPdf(string path) =>
+        string.Equals(System.IO.Path.GetExtension(path), ".pdf", StringComparison.OrdinalIgnoreCase);

+    /// <summary>
+    /// Ouvre un PDF dans un nouvel onglet. Demande le mot de passe si le fichier est protégé.
+    /// Retourne null si l'ouverture est annulée ou impossible (le message a déjà été affiché).
+    /// </summary>
+    public static async Task<PdfViewer?> OpenAsync(string path, Window owner)
+    {
+        string fullPath = System.IO.Path.GetFullPath(path);
+        string workFolder = System.IO.Path.Combine(System.IO.Path.GetTempPath(), "KEYCE_KODER", "pdf", Guid.NewGuid().ToString("N"));
+
+        try
+        {
+            Directory.CreateDirectory(workFolder);
+            string firstCopy = System.IO.Path.Combine(workFolder, "v0.pdf");
+            File.Copy(fullPath, firstCopy, overwrite: true);
+
+            string? password = null;
+
+            for (int attempt = 0; ; attempt++)
+            {
+                try
+                {
+                    PdfRenderer renderer = await PdfRenderer.LoadAsync(firstCopy, password);
+                    return new PdfViewer(fullPath, workFolder, firstCopy, password, renderer);
+                }
+                catch (Exception ex) when (PdfRenderer.IsPasswordError(ex) && attempt < 3)
+                {
+                    string[]? values = PromptWindow.Ask(
+                        owner,
+                        "PDF protégé",
+                        attempt == 0
+                            ? $"« {System.IO.Path.GetFileName(fullPath)} » est protégé par un mot de passe."
+                            : "Mot de passe incorrect, réessayez.",
+                        ("Mot de passe", ""));
+
+                    if (values is null)
+                    {
+                        TryDeleteFolder(workFolder);
+                        return null;
+                    }
+
+                    password = values[0];
+                }
+            }
+        }
+        catch (Exception ex)
+        {
+            TryDeleteFolder(workFolder);
+            MessageBox.Show(owner,
+                $"Impossible d'ouvrir ce PDF :\n{fullPath}\n\n{ex.Message}",
+                "Lecteur PDF", MessageBoxButton.OK, MessageBoxImage.Error);
+            return null;
+        }
+    }

+    private async Task ReloadAsync(int page)
+    {
+        _page = Math.Clamp(page, 0, _renderer.PageCount - 1);
+        _pageCount.Text = $"/ {_renderer.PageCount}";
+        await RenderPageAsync();
+        _ = RenderThumbnailsAsync();
+    }
+
+    private async Task RenderPageAsync()
+    {
+        int ticket = ++_renderTicket;
+
+        try
+        {
+            Size size = _renderer.GetPageSize(_page);
+            double width = size.Width * _zoom;
+            double height = size.Height * _zoom;
+            double dpi = VisualTreeHelper.GetDpi(this).DpiScaleX;
+
+            BitmapSource bitmap = await _renderer.RenderAsync(_page, width * dpi);
+
+            // Un rendu plus récent a été demandé entre-temps : on ignore celui-ci
+            if (ticket != _renderTicket)
+                return;
+
+            _pageImage.Source = bitmap;
+            _pageImage.Width = width;
+            _pageImage.Height = height;
+            _overlay.Width = width;
+            _overlay.Height = height;
+
+            _pageBox.Text = (_page + 1).ToString();
+            _zoomText.Text = $"{Math.Round(_zoom * 100)} %";
+
+            if (_thumbnails.SelectedIndex != _page && _page < _thumbnails.Items.Count)
+                _thumbnails.SelectedIndex = _page;
+
+            if (_textPanel.Visibility == Visibility.Visible)
+                await LoadPageTextAsync();
+        }
+        catch (Exception ex)
+        {
+            ShowStatus($"❌ Affichage impossible : {ex.Message}", error: true);
+        }
+
+        UpdateButtons();
+    }
+
+    private async Task RenderThumbnailsAsync()
+    {
+        int ticket = ++_thumbnailTicket;
+        int count = _renderer.PageCount;
+        double dpi = VisualTreeHelper.GetDpi(this).DpiScaleX;
+
+        _thumbnails.Items.Clear();
+        var images = new List<Image>();
+
+        for (int i = 0; i < count; i++)
+        {
+            var image = new Image { Width = ThumbnailWidth, Stretch = Stretch.Uniform };
+            var item = new StackPanel { Margin = new Thickness(0, 6, 0, 2) };
+            item.Children.Add(new Border { Background = Brushes.White, Child = image });
+            item.Children.Add(new TextBlock
+            {
+                Text = (i + 1).ToString(),
+                HorizontalAlignment = HorizontalAlignment.Center,
+                Foreground = Theme.Muted,
+                Margin = new Thickness(0, 3, 0, 0)
+            });
+
+            images.Add(image);
+            _thumbnails.Items.Add(item);
+        }
+
+        _thumbnails.SelectedIndex = _page;
+
+        // Les miniatures sont dessinées une par une, sans bloquer l'interface
+        for (int i = 0; i < count; i++)
+        {
+            if (ticket != _thumbnailTicket)
+                return; // document rechargé entre-temps
+
+            try
+            {
+                images[i].Source = await _renderer.RenderAsync(i, ThumbnailWidth * dpi);
+            }
+            catch (Exception)
+            {
+                // Miniature impossible : on laisse la case blanche
+            }
+        }
+    }
+
+    private async Task GoToPageAsync(int page)
+    {
+        if (page < 0 || page >= _renderer.PageCount || page == _page && _pageImage.Source is not null)
+        {
+            _pageBox.Text = (_page + 1).ToString();
+            return;
+        }
+
+        _page = page;
+        await RenderPageAsync();
+        _scroll.ScrollToTop();
+    }
+
+    private async Task SetZoomAsync(double zoom)
+    {
+        _zoom = Math.Clamp(zoom, MinZoom, MaxZoom);
+        await RenderPageAsync();
+    }
+
+    private Task FitWidthAsync()
+    {
+        Size size = _renderer.GetPageSize(_page);
+        double available = _scroll.ViewportWidth - 40;
+        return available > 50 ? SetZoomAsync(available / size.Width) : Task.CompletedTask;
+    }
+
+    private async void OnMouseWheel(object sender, MouseWheelEventArgs e)
+    {
+        // Ctrl+molette = zoom
+        if (Keyboard.Modifiers == ModifierKeys.Control)
+        {
+            e.Handled = true;
+            await SetZoomAsync(e.Delta > 0 ? _zoom * 1.1 : _zoom / 1.1);
+            return;
+        }
+
+        // En bas de la page, la molette passe à la page suivante (et inversement)
+        if (e.Delta < 0 && _scroll.VerticalOffset >= _scroll.ScrollableHeight && _page < _renderer.PageCount - 1)
+        {
+            e.Handled = true;
+            await GoToPageAsync(_page + 1);
+        }
+        else if (e.Delta > 0 && _scroll.VerticalOffset <= 0 && _page > 0)
+        {
+            e.Handled = true;
+            await GoToPageAsync(_page - 1);
+            _scroll.ScrollToBottom();
+        }
+    }

+    /// <summary>
+    /// Applique une modification sur une nouvelle copie, puis l'affiche.
+    /// L'ancienne copie est gardée pour pouvoir annuler.
+    /// </summary>
+    private async Task EditAsync(string description, PdfEdit edit, int pageAfter)
+    {
+        if (_busy)
+            return;
+
+        SetBusy(true);
+        ShowStatus("Modification en cours…");
+
+        string next = System.IO.Path.Combine(_workFolder, $"v{_history.Count + 1}-{Guid.NewGuid():N}.pdf");
+
+        try
+        {
+            await PdfEditor.ApplyAsync(_workingPath, next, _password, edit);
+            PdfRenderer renderer = await PdfRenderer.LoadAsync(next, _password);
+
+            _history.Push(_workingPath);
+            _workingPath = next;
+            _renderer = renderer;
+
+            await ReloadAsync(pageAfter);
+            ShowStatus($"✓ {description}. (Ctrl+Z pour annuler, « Enregistrer » pour garder)");
+        }
+        catch (Exception ex)
+        {
+            TryDelete(next);
+            ShowStatus($"❌ {ex.Message}", error: true);
+        }
+        finally
+        {
+            SetBusy(false);
+            StateChanged?.Invoke(this);
+        }
+    }

+    private Task MovePageUpAsync() =>
+        _page == 0 ? Task.CompletedTask : EditAsync("Page déplacée", PdfEditor.MovePage(_page, _page - 1), _page - 1);
+
+    private Task MovePageDownAsync() =>
+        _page >= _renderer.PageCount - 1 ? Task.CompletedTask : EditAsync("Page déplacée", PdfEditor.MovePage(_page, _page + 1), _page + 1);
+
+    private Task DeletePageAsync()
+    {
+        MessageBoxResult answer = MessageBox.Show(
+            Window.GetWindow(this)!,
+            $"Supprimer la page {_page + 1} ?\n(Vous pourrez annuler avec Ctrl+Z tant que vous n'avez pas enregistré.)",
+            "Lecteur PDF", MessageBoxButton.YesNo, MessageBoxImage.Question);
+
+        return answer == MessageBoxResult.Yes
+            ? EditAsync("Page supprimée", PdfEditor.DeletePage(_page), Math.Max(0, _page - 1))
+            : Task.CompletedTask;
+    }

+    private Task MergeAsync()
+    {
+        var dialog = new OpenFileDialog
+        {
+            Title = "PDF à ajouter à la fin du document",
+            Filter = "Documents PDF (*.pdf)|*.pdf"
+        };
+
+        if (dialog.ShowDialog(Window.GetWindow(this)) != true)
+            return Task.CompletedTask;
+
+        return EditAsync($"Pages de « {System.IO.Path.GetFileName(dialog.FileName)} » ajoutées",
+            PdfEditor.AppendDocument(dialog.FileName), _renderer.PageCount);
+    }

+    private async Task UndoAsync()
+    {
+        if (_busy || _history.Count == 0)
+            return;
+
+        SetBusy(true);
+
+        try
+        {
+            string previous = _history.Pop();
+            _renderer = await PdfRenderer.LoadAsync(previous, _password);
+            _workingPath = previous;
+
+            await ReloadAsync(_page);
+            ShowStatus("↶ Modification annulée.");
+        }
+        catch (Exception ex)
+        {
+            ShowStatus($"❌ {ex.Message}", error: true);
+        }
+        finally
+        {
+            SetBusy(false);
+            StateChanged?.Invoke(this);
+        }
+    }

+    private void SelectTool(Tool tool)
+    {
+        _tool = _tool == tool ? Tool.None : tool;
+
+        foreach ((Tool key, ToggleButton button) in _toolButtons)
+            button.IsChecked = key == _tool;
+
+        _overlay.Cursor = _tool == Tool.None ? null : Cursors.Cross;
+
+        ShowStatus(_tool switch
+        {
+            Tool.Text => "Cliquez sur la page à l'endroit où écrire le texte (Échap pour arrêter).",
+            Tool.Highlight => "Tracez un rectangle sur la zone à surligner (Échap pour arrêter).",
+            Tool.Mask => "Tracez un rectangle sur la zone à masquer en blanc (Échap pour arrêter).",
+            _ => ""
+        });
+    }

+    private async void OnOverlayMouseDown(object sender, MouseButtonEventArgs e)
+    {
+        if (_tool == Tool.None || _busy || _overlay.ActualWidth <= 0)
+            return;
+
+        Point position = e.GetPosition(_overlay);
+        e.Handled = true;
+
+        if (_tool == Tool.Text)
+        {
+            await AddTextAtAsync(position);
+            return;
+        }
+
+        // Surligner / masquer : début du rectangle
+        _dragStart = position;
+        _dragging = true;
+        _overlay.CaptureMouse();
+
+        _selection.Fill = _tool == Tool.Highlight ? HighlightFill : MaskFill;
+        _selection.Stroke = Theme.Accent;
+        Canvas.SetLeft(_selection, position.X);
+        Canvas.SetTop(_selection, position.Y);
+        _selection.Width = 0;
+        _selection.Height = 0;
+        _selection.Visibility = Visibility.Visible;
+    }
+
+    private void OnOverlayMouseMove(object sender, MouseEventArgs e)
+    {
+        if (!_dragging)
+            return;
+
+        Point position = e.GetPosition(_overlay);
+        double x = Math.Clamp(Math.Min(position.X, _dragStart.X), 0, _overlay.ActualWidth);
+        double y = Math.Clamp(Math.Min(position.Y, _dragStart.Y), 0, _overlay.ActualHeight);
+        double right = Math.Clamp(Math.Max(position.X, _dragStart.X), 0, _overlay.ActualWidth);
+        double bottom = Math.Clamp(Math.Max(position.Y, _dragStart.Y), 0, _overlay.ActualHeight);
+
+        Canvas.SetLeft(_selection, x);
+        Canvas.SetTop(_selection, y);
+        _selection.Width = right - x;
+        _selection.Height = bottom - y;
+    }
+
+    private async void OnOverlayMouseUp(object sender, MouseButtonEventArgs e)
+    {
+        if (!_dragging)
+            return;
+
+        _dragging = false;
+        _overlay.ReleaseMouseCapture();
+        _selection.Visibility = Visibility.Collapsed;
+
+        double width = _overlay.ActualWidth;
+        double height = _overlay.ActualHeight;
+        double fx = Canvas.GetLeft(_selection) / width;
+        double fy = Canvas.GetTop(_selection) / height;
+        double fw = _selection.Width / width;
+        double fh = _selection.Height / height;
+
+        if (_selection.Width < 4 || _selection.Height < 4)
+            return; // simple clic : rien à faire
+
+        if (_tool == Tool.Highlight)
+            await EditAsync("Zone surlignée", PdfEditor.Highlight(_page, fx, fy, fw, fh), _page);
+        else if (_tool == Tool.Mask)
+            await EditAsync("Zone masquée", PdfEditor.Mask(_page, fx, fy, fw, fh), _page);
+    }
+
+    private async Task AddTextAtAsync(Point position)
+    {
+        string[]? values = PromptWindow.Ask(
+            Window.GetWindow(this),
+            "Ajouter du texte",
+            "Texte à écrire sur la page (le coin haut-gauche du texte sera à l'endroit cliqué).\n"
+            + "Pour plusieurs lignes, séparez-les par « \\n ».",
+            ("Texte", ""),
+            ("Taille (points)", "12"),
+            ("Couleur : noir, rouge ou bleu", "noir"));
+
+        if (values is null || values[0].Length == 0)
+            return;
+
+        if (!double.TryParse(values[1].Replace(',', '.'), System.Globalization.NumberStyles.Float,
+                System.Globalization.CultureInfo.InvariantCulture, out double size) || size < 4 || size > 200)
+            size = 12;
+

+    /// <summary>Relit l'état du dépôt et met la vue à jour.</summary>
+    public async Task RefreshAsync()
+    {
+        // Un rafraîchissement est déjà en cours : on en refera un juste après

+    private async Task RefreshCoreAsync()
+    {
+        string? folder = _getFolder();
+
+        if (folder is null || !Directory.Exists(folder))
+        {
+            SetRepository(null);
+            ShowState(
+                "Aucun dossier ouvert",
+                "Ouvrez le dossier de votre projet pour utiliser Git, ou clonez un dépôt GitHub.",
+                ("Ouvrir un dossier", () =>
+                {
+                    ChooseFolderRequested?.Invoke();
+                    return Task.CompletedTask;
+                }),
+                ("Cloner un dépôt…", CloneAsync));
+            return;
+        }
+
+        if (!_gitFound)
+        {
+            _gitFound = await _git.IsInstalledAsync();
+
+            if (!_gitFound)
+            {
+                SetRepository(null);
+                await ShowGitMissingAsync();
+                return;
+            }
+        }
+
+        string? root = await _git.FindRepositoryRootAsync(folder);
+
+        if (root is null)
+        {
+            SetRepository(null);
+            ShowState(
+                "Pas encore de dépôt Git",
+                $"Le dossier « {Path.GetFileName(Path.TrimEndingDirectorySeparator(folder))} » n'est pas suivi par Git.\n"
+                + "Initialisez un dépôt pour commencer à enregistrer l'historique de vos modifications.",
+                ("Initialiser le dépôt", () => InitAsync(folder)),
+                ("Cloner un dépôt…", CloneAsync));
+            return;
+        }
+
+        GitStatus status;
+        IReadOnlyList<GitCommitInfo> history;
+
+        try
+        {
+            status = await _git.GetStatusAsync(root);
+            history = await _git.GetHistoryAsync(root, 15);
+        }
+        catch (GitException ex)
+        {
+            SetRepository(null);
+            ShowState("Erreur Git", ex.Message);
+            return;
+        }

+        SetRepository(root);
+        _status = status;

+                stage.Click += async (_, _) => await StageAsync(change);
+                actions.Children.Add(stage);
+                break;
+
+            case SectionKind.Staged:
+                Button unstage = SmallButton("−", "Désindexer");
+                unstage.Click += async (_, _) => await UnstageAsync(change);
+                actions.Children.Add(unstage);
+                break;
+
+            case SectionKind.Conflicts:
+                Button resolve = SmallButton("+", "Marquer comme résolu");
+                resolve.Click += async (_, _) => await StageAsync(change);
+                actions.Children.Add(resolve);
+                break;
+        }
+
+        Grid.SetColumn(badge, 0);
+        Grid.SetColumn(name, 1);
+        Grid.SetColumn(actions, 2);
+        row.Children.Add(badge);
+        row.Children.Add(name);
+        row.Children.Add(actions);
+
+        row.MouseEnter += (_, _) =>
+        {
+            row.Background = Theme.Surface;
+            actions.Visibility = Visibility.Visible;
+        };
+        row.MouseLeave += (_, _) =>
+        {
+            row.Background = Brushes.Transparent;
+            actions.Visibility = Visibility.Hidden;
+        };
+
+        // Double-clic = voir les modifications
+        row.MouseLeftButtonDown += async (_, e) =>
+        {
+            if (e.ClickCount == 2)
+            {
+                e.Handled = true;
+                await ShowDiffAsync(change);
+            }
+        };
+
+        string renamed = change.OriginalPath is null ? "" : $"\nRenommé depuis : {change.OriginalPath}";
+        row.ToolTip = $"{change.Path}{renamed}\n{change.Description}"
+                      + (kind == SectionKind.Staged ? " — indexé" : "")
+                      + "\nDouble-clic : voir les modifications";
+
+        row.ContextMenu = BuildContextMenu(change, kind);
+        return row;
+    }

+    {
+        var menu = new ContextMenu();

+        menu.Items.Add(MenuEntry("Voir les modifications", () => ShowDiffAsync(change)));

+        if (change.Kind != GitChangeKind.Deleted)

+                return Task.CompletedTask;

+        {

+            case SectionKind.Staged:
+                menu.Items.Add(MenuEntry("Désindexer (−)", () => UnstageAsync(change)));
+                break;

+            case SectionKind.Conflicts:
+                menu.Items.Add(MenuEntry("Marquer comme résolu (+)", () => StageAsync(change)));

+    private async Task CommitAsync()
+    {
+        if (_root is null || _status is null || _busy)
+            return;
+
+        string root = _root;
+        GitStatus status = _status;
+        string message = _message.Text.Trim();
+
+        if (status.HasConflicts)
+        {
+            ShowInfo("Résolvez d'abord les conflits, puis marquez chaque fichier comme résolu avec +.", error: true);
+            return;
+        }
+
+        if (status.IsClean)
+        {
+            ShowInfo("Rien à valider : aucune modification.", error: false);
+            return;
+        }
+
+        if (message.Length == 0)
+        {
+            ShowInfo("Écrivez d'abord un message qui décrit vos modifications (ex. « Ajout du menu Fichier »).", error: true);
+            _message.Focus();
+            return;
+        }
+
+        // Rien d'indexé : comme VS Code, on propose de tout valider
+        bool stageAll = false;
+        if (!status.HasStaged)
+        {
+            MessageBoxResult answer = MessageBox.Show(
+                Owner(),
+                "Aucune modification n'est indexée.\n\nVoulez-vous indexer toutes les modifications et les valider ?",
+                "Git",
+                MessageBoxButton.YesNo,
+                MessageBoxImage.Question);
+
+            if (answer != MessageBoxResult.Yes)
+                return;
+
+            stageAll = true;
+        }
+
+        if (!await EnsureIdentityAsync(root))
+            return;
+
+        bool success = await ExecuteAsync("Validation (commit)…", "✅ Commit effectué. Cliquez sur « Push » pour l'envoyer sur GitHub.",
+            async () =>
+            {
+                if (stageAll)
+                {
+                    GitResult staged = await _git.StageAllAsync(root);
+                    if (!staged.Success)
+                        return staged;
+                }
+
+                return await _git.CommitAsync(root, message);
+            });
+
+        if (success)
+            _message.Clear();
+    }
+
+    /// <summary>Git exige un nom et un e-mail pour signer les commits : on les demande une seule fois.</summary>
+    private async Task<bool> EnsureIdentityAsync(string root)
+    {
+        if (await _git.HasIdentityAsync(root))
+            return true;
+
+        string[]? values = PromptWindow.Ask(
+            Owner(),
+            "Identité Git",
+            "Git a besoin de votre nom et de votre e-mail pour signer vos commits.\n"
+            + "Utilisez de préférence l'e-mail de votre compte GitHub. (Demandé une seule fois sur ce PC.)",
+            ("Nom", ""),
+            ("E-mail", ""));
+
+        if (values is null)
+            return false;
+
+        if (values[0].Length == 0 || !values[1].Contains('@'))
+        {
+            ShowInfo("Nom ou e-mail invalide : le commit n'a pas été fait.", error: true);
+            return false;
+        }
+
+        GitResult result = await _git.SetIdentityAsync(root, values[0], values[1]);
+        if (!result.Success)
+        {
+            ShowInfo("❌ " + Shorten(result.Message), error: true);
+            return false;
+        }
+
+        return true;
+    }

+    private async Task PushAsync()
+    {
+        if (_root is null || _status is null || _busy)
+            return;
+
+        string root = _root;
+        GitStatus status = _status;
+
+        if (status.IsDetached)
+        {
+            ShowInfo("Impossible de pousser : aucune branche active (HEAD détaché).", error: true);
+            return;
+        }
+
+        if (!status.HasCommits)
+        {
+            ShowInfo("Faites d'abord un commit avant de pousser.", error: true);
+            return;
+        }
+
+        // Première publication de la branche : il faut un dépôt distant (origin)
+        string? remote = null;
+        if (status.Upstream is null)
+        {
+            remote = await EnsureRemoteAsync(root);
+            if (remote is null)
+                return;
+        }
+
+        string branch = status.Branch!;
+        await ExecuteAsync(
+            "Push en cours… (une fenêtre de connexion à GitHub peut s'ouvrir la première fois)",
+            "✅ Push terminé : vos commits sont sur GitHub.",
+            () => _git.PushAsync(root, branch, remote));
+    }

+    /// <summary>Nom du dépôt distant à utiliser ; s'il n'y en a pas, demande l'adresse et ajoute « origin ».</summary>
+    private async Task<string?> EnsureRemoteAsync(string root)
+    {
+        IReadOnlyList<string> remotes = await _git.GetRemotesAsync(root);
+
+        if (remotes.Contains("origin"))
+            return "origin";
+
+        if (remotes.Count > 0)
+            return remotes[0];
+
+        string[]? values = PromptWindow.Ask(
+            Owner(),
+            "Dépôt distant",
+            "Ce projet n'est relié à aucun dépôt en ligne.\n"
+            + "Créez un dépôt vide sur GitHub, puis collez son adresse ici "
+            + "(ex. https://github.com/nom/projet.git).",
+            ("Adresse du dépôt", ""));
+
+        if (values is null || values[0].Length == 0)
+            return null;
+
+        GitResult result = await _git.AddRemoteAsync(root, "origin", values[0]);
+        if (!result.Success)
+        {
+            ShowInfo("❌ " + Shorten(result.Message), error: true);
+            return null;
+        }
+
+        return "origin";
+    }

+    private async Task InstallGitAsync()
+    {
+        if (_busy)
+            return;
+
+        SetBusy(true);
+        ShowInfo("Installation de Git avec winget… Acceptez la demande d'autorisation de Windows si elle apparaît.", error: false);
+
+        try
+        {
+            int exitCode = await new ToolInstaller().InstallAsync(
+                GitInstall,
+                line => Dispatcher.BeginInvoke(new Action(() => ShowInfo(line, error: false))));
+
+            // Git vient d'être installé : on relit le PATH (pas besoin de redémarrer)
+            EnvironmentPath.Refresh();
+
+            if (!await _git.IsInstalledAsync())
+            {
+                string? folder = EnvironmentPath.FindExecutableFolder(GitInstall);
+                if (folder is not null)
+                    EnvironmentPath.AddToUserPath(folder);
+            }
+
+            _gitFound = await _git.IsInstalledAsync();
+
+            ShowInfo(_gitFound
+                    ? "✅ Git est installé."
+                    : $"⚠ Git n'est pas encore détecté (code {exitCode}). Redémarrez KEYCE KODER.",
+                error: !_gitFound);
+        }
+        catch (Exception ex)
+        {
+            ShowInfo("❌ Installation impossible : " + ex.Message, error: true);
+        }
+        finally
+        {
+            SetBusy(false);
+        }
+
+        await RefreshAsync();
+    }

+    private static string RepositoryNameFromUrl(string url)
+    {
+        string name = url.TrimEnd('/', '\\');
+        int separator = name.LastIndexOfAny(['/', '\\', ':']);
+        if (separator >= 0)
+            name = name[(separator + 1)..];
+
+        if (name.EndsWith(".git", StringComparison.OrdinalIgnoreCase))
+            name = name[..^4];
+
+        foreach (char invalid in Path.GetInvalidFileNameChars())
+            name = name.Replace(invalid, '_');
+
+        return name.Length == 0 ? "depot" : name;
+    }

+    /// <summary>
+    /// Lance une opération Git : bloque la vue pendant l'opération, affiche le résultat,
+    /// puis relit l'état du dépôt. Retourne true si l'opération a réussi.
+    /// </summary>
+    private async Task<bool> ExecuteAsync(
+        string runningText,
+        string? successText,
+        Func<Task<GitResult>> operation,
+        bool changesFiles = false)
+    {
+        if (_busy)
+            return false;
+
+        SetBusy(true);
+        ShowInfo(runningText, error: false);
+
+        GitResult result;
+
+        try
+        {
+            result = await operation();
+        }
+        catch (Exception ex)
+        {
+            result = GitResult.Fail(ex.Message);
+        }
+        finally
+        {
+            SetBusy(false);
+        }
+
+        // Pull, changement de branche, annulation : les onglets ouverts doivent être rechargés
+        if (changesFiles)
+            WorkingTreeChanged?.Invoke();
+
+        await RefreshAsync();
+
+        if (result.Success)
+            ShowInfo(successText ?? string.Empty, error: false);
+        else
+            ShowInfo("❌ " + Shorten(result.Message), error: true, fullText: result.Message);
+
+        return result.Success;
+    }

+    /// <summary>Garde les dernières lignes d'un message Git (souvent long).</summary>
+    private static string Shorten(string message)
+    {
+        string[] lines = message
+            .Replace("\r\n", "\n")
+            .Split('\n', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries);
+
+        string text = string.Join("\n", lines.TakeLast(6));
+        if (text.Length == 0)
+            text = "La commande Git a échoué.";
+
+        return text.Length > 500 ? text[..500] + "…" : text;
+    }

+    // =====================================================================
+    // Surveillance des fichiers : la vue se met à jour toute seule
+    // =====================================================================
+
+    private void Watch(string? root)
+    {
+        if (string.Equals(root, _watchedRoot, StringComparison.OrdinalIgnoreCase))
+            return;
+
+        _watcher?.Dispose();
+        _watcher = null;
+        _watchedRoot = root;
+
+        if (root is null)
+            return;
+
+        try
+        {
+            var watcher = new FileSystemWatcher(root)
+            {
+                IncludeSubdirectories = true,
+                NotifyFilter = NotifyFilters.FileName | NotifyFilters.DirectoryName | NotifyFilters.LastWrite
+            };
+
+            watcher.Changed += OnFileSystemChanged;
+            watcher.Created += OnFileSystemChanged;
+            watcher.Deleted += OnFileSystemChanged;
+            watcher.Renamed += OnFileSystemChanged;
+            watcher.EnableRaisingEvents = true;
+
+            _watcher = watcher;
+        }
+        catch (Exception ex) when (ex is ArgumentException or IOException or PlatformNotSupportedException)
+        {
+            // Pas de surveillance possible : le bouton ⟳ reste disponible
+        }
+    }
+
+    // Appelé sur un thread de fond : on repasse sur le thread de l'interface
+    private void OnFileSystemChanged(object sender, FileSystemEventArgs e)
+    {
+        if (IsRelevant(e.FullPath))
+            Dispatcher.BeginInvoke(new Action(ScheduleRefresh));
+    }
+
+    private void ScheduleRefresh()
+    {
+        // Vue cachée : elle sera relue quand elle redeviendra visible
+        if (!IsVisible)
+            return;
+
+        // On redémarre le minuteur : un seul rafraîchissement après une série de changements
+        _refreshTimer.Stop();
+        _refreshTimer.Start();
+    }
+
+    private static bool IsRelevant(string fullPath)
+    {
+        string[] parts = fullPath.Split(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar);
+        string name = parts[^1];
+
+        if (name.EndsWith(".lock", StringComparison.OrdinalIgnoreCase))
+            return false;
+
+        int gitIndex = Array.FindIndex(parts, part => part.Equals(".git", StringComparison.OrdinalIgnoreCase));
+        if (gitIndex >= 0)
+        {
+            // Dans .git : seuls l'index, HEAD et les références comptent (commit, branche, pull...)
+            return name is "index" or "HEAD"
+                   || (gitIndex + 1 < parts.Length && parts[gitIndex + 1] == "refs");
+        }
+
+        return !parts.Any(part =>
+            part.Equals(".vs", StringComparison.OrdinalIgnoreCase)
+            || part.Equals("node_modules", StringComparison.OrdinalIgnoreCase));
+    }

+/// <summary>
+/// Lance les commandes git (le programme git.exe installé sur le PC) et renvoie leurs résultats.
+/// Aucune dépendance à l'interface : utilisable et testable partout.
+/// </summary>
+public sealed class GitService
+{
+    // Options ajoutées à chaque commande :
+    // - core.quotepath=false : accents affichés normalement (pas « \303\251 »)
+    // - color.ui=never       : pas de codes couleur dans la sortie
+    private static readonly string[] CommonOptions =
+    [
+        "-c", "core.quotepath=false",
+        "-c", "color.ui=never"
+    ];
+
+    private static readonly UTF8Encoding Utf8 = new(encoderShouldEmitUTF8Identifier: false);
+
+    // ---------------- Exécution ----------------
+
+    /// <summary>Lance « git arguments... » dans le dossier indiqué.</summary>
+    public async Task<GitResult> RunAsync(
+        string workingDirectory,
+        IEnumerable<string> arguments,
+        CancellationToken cancellationToken = default)
+    {
+        if (!Directory.Exists(workingDirectory))
+            return GitResult.Fail($"Le dossier n'existe pas : {workingDirectory}");
+
+        var startInfo = new ProcessStartInfo("git")
+        {
+            WorkingDirectory = workingDirectory,
+            UseShellExecute = false,
+            CreateNoWindow = true,
+            RedirectStandardOutput = true,
+            RedirectStandardError = true,
+            StandardOutputEncoding = Utf8,
+            StandardErrorEncoding = Utf8
+        };
+
+        foreach (string option in CommonOptions)
+            startInfo.ArgumentList.Add(option);
+
+        foreach (string argument in arguments)
+            startInfo.ArgumentList.Add(argument);
+
+        // Jamais de question dans une console invisible (sinon git attendrait indéfiniment)
+        startInfo.Environment["GIT_TERMINAL_PROMPT"] = "0";
+        // « git status » ne réécrit pas l'index (évite des rafraîchissements en boucle)
+        startInfo.Environment["GIT_OPTIONAL_LOCKS"] = "0";
+        // Les chemins sont pris à la lettre (un fichier nommé « *.txt » reste « *.txt »)
+        startInfo.Environment["GIT_LITERAL_PATHSPECS"] = "1";
+
+        using var process = new Process { StartInfo = startInfo };
+
+        try
+        {
+            process.Start();
+        }
+        catch (Win32Exception)
+        {
+            return new GitResult(-1, string.Empty, "Git est introuvable sur ce PC.");
+        }
+
+        Task<string> output = process.StandardOutput.ReadToEndAsync(cancellationToken);
+        Task<string> error = process.StandardError.ReadToEndAsync(cancellationToken);
+
+        try
+        {
+            await process.WaitForExitAsync(cancellationToken);
+        }
+        catch (OperationCanceledException)
+        {
+            try { process.Kill(entireProcessTree: true); }
+            catch (InvalidOperationException) { /* déjà terminé */ }
+            throw;
+        }
+
+        return new GitResult(process.ExitCode, await output, await error);
+    }
+

+    /// <summary>Différences d'un fichier (format « diff » : lignes + ajoutées, - supprimées).</summary>

+        if (change.Kind == GitChangeKind.Untracked)
+            return await DescribeNewFileAsync(root, change);
+
+        // Indexé : différence index ↔ dernier commit ; non indexé : fichier ↔ index
+        var arguments = new List<string> { "diff" };
+
+        if (change.Staged)
+        {
+            arguments.Add("--cached");
+            arguments.Add("-M");
+        }
+
+        arguments.Add("--");
+
+        if (change.Staged && change.OriginalPath is not null)
+            arguments.Add(change.OriginalPath);
+
+        arguments.Add(change.Path);
+
+        GitResult result = await RunAsync(root, arguments);
+        return result.Success ? result.Output : result.Message;
+    }

+    /// </summary>
+    public async Task<GitResult> DiscardAsync(string root, GitFileChange change)
+    {
+        if (change.Kind != GitChangeKind.Untracked)
+            return await RunAsync(root, ["restore", "--", change.Path]);

+        string fullPath = ToFullPath(root, change.Path);

+        {
+            return GitResult.Fail(ex.Message);
+        }
+    }

+    private static async Task<string> DescribeNewFileAsync(string root, GitFileChange change)
+    {
+        string fullPath = ToFullPath(root, change.Path);
+
+        try
+        {
+            var info = new FileInfo(fullPath);
+            if (!info.Exists)
+                return "(Fichier introuvable.)";
+
+            if (info.Length > 1_000_000)
+                return "(Fichier trop volumineux pour être affiché.)";
+
+            string text = await File.ReadAllTextAsync(fullPath);
+            if (text.Contains('\0'))
+                return "(Fichier binaire : aperçu impossible.)";
+
+            var builder = new StringBuilder();
+            builder.Append("--- /dev/null\n");
+            builder.Append("+++ b/").Append(change.Path).Append('\n');
+
+            foreach (string line in text.Replace("\r\n", "\n").Split('\n'))
+                builder.Append('+').Append(line).Append('\n');
+
+            return builder.ToString();
+        }
+        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
+        {
+            return $"(Lecture impossible : {ex.Message})";
+        }
+    }
